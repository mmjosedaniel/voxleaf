#!/usr/bin/env bash
set -euo pipefail
root=/var/tmp/voxleaf-linux-rust-comparison-20261010
repo=/mnt/c/Users/mmjos/Desktop/workbeanch/voxleaf
attempt="$root/complete-02"
test ! -e "$attempt"
mkdir -p "$attempt/baseline" "$attempt/evidence"
tar -xf "$repo/tmp/linux-rust-comparison/baseline-full.tar" -C "$attempt/baseline"
cp -a "$attempt/baseline" "$attempt/current"
cp -a "$repo/apps/desktop/src-tauri/src/." "$attempt/current/apps/desktop/src-tauri/src/"
export CARGO_HOME="$root/tools/cargo"
export RUSTUP_HOME="$root/tools/rustup"
export CARGO_TARGET_DIR="$root/target"
export CARGO_BUILD_JOBS=2
export PATH="$root/tools/pnpm/bin:$root/tools/node-v24.18.0-linux-x64/bin:$CARGO_HOME/bin:/usr/local/bin:/usr/bin:/bin"
printf 'snapshot\tcommand\texit\n' > "$attempt/evidence/results.tsv"
overall=0
for snapshot in baseline current; do
  cd "$attempt/$snapshot"
  find . -type f -print0 | LC_ALL=C sort -z | xargs -0 sha256sum > "$attempt/evidence/$snapshot-inputs.sha256"
  pnpm install --frozen-lockfile --ignore-scripts > "$attempt/evidence/$snapshot-bootstrap.log" 2>&1
  for script in format:check:rust lint:rust test:rust; do
    logfile="$attempt/evidence/$snapshot-${script//:/-}.log"
    printf '\nRunning %s: pnpm %s\n' "$snapshot" "$script"
    set +e
    pnpm "$script" > "$logfile" 2>&1
    result=$?
    set -e
    printf '%s\t%s\t%s\n' "$snapshot" "$script" "$result" >> "$attempt/evidence/results.tsv"
    tail -n 22 "$logfile"
    printf 'Exit: %s\n' "$result"
    if [ "$result" -ne 0 ]; then overall=1; fi
  done
  sha256sum --check "$attempt/evidence/$snapshot-inputs.sha256" > "$attempt/evidence/$snapshot-input-recheck.log" 2>&1
  find . -type f -not -path '*/node_modules/*' -not -path './apps/desktop/src-tauri/gen/*' -print | LC_ALL=C sort > "$attempt/evidence/$snapshot-final-paths.txt"
done
cat "$attempt/evidence/results.tsv"
exit "$overall"
