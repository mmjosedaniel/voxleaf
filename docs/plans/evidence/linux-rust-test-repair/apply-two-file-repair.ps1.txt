function Crc32([byte[]]$bytes, [int]$offset, [int]$length) {
    [uint64]$crc = 4294967295
    for ($i=$offset; $i -lt $offset+$length; $i++) {
        $crc = $crc -bxor [uint64]$bytes[$i]
        for ($j=0; $j -lt 8; $j++) {
            if ($crc -band 1) { $crc = ($crc -shr 1) -bxor [uint64]3988292384 } else { $crc = $crc -shr 1 }
        }
    }
    [uint32]($crc -bxor [uint64]4294967295)
}
