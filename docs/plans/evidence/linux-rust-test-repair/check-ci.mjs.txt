import fs from 'node:fs';
import path from 'node:path';
import assert from 'node:assert/strict';
import { createRequire } from 'node:module';
const require = createRequire(import.meta.url);
const yaml = require(path.resolve('node_modules/.pnpm/js-yaml@4.3.0/node_modules/js-yaml'));
const before = yaml.load(fs.readFileSync('tmp/linux-rust-repair/workflow-before.yml', 'utf8'));
const current = yaml.load(fs.readFileSync('.github/workflows/foundation-checks.yml', 'utf8'));
const { jobs: beforeJobs, ...beforeRoot } = before;
const { jobs, ...currentRoot } = current;
assert.deepEqual(currentRoot, beforeRoot);
for (const [name, job] of Object.entries(beforeJobs)) assert.deepEqual(jobs[name], job);
assert.deepEqual(Object.keys(jobs).sort(), [...Object.keys(beforeJobs), 'ubuntu-native'].sort());
const job = jobs['ubuntu-native'];
assert.equal(job['runs-on'], 'ubuntu-24.04');
assert.equal(job['timeout-minutes'], 45);
assert.equal(job.defaults.run.shell, 'bash');
assert.equal(String(job.env.CARGO_BUILD_JOBS), '2');
const sourceSteps = beforeJobs['ubuntu-portable'].steps;
for (const step of job.steps.filter(step => step.uses)) {
  const source = sourceSteps.find(source => source.uses === step.uses);
  assert.ok(source, 'No new/unpinned action');
  assert.deepEqual(step.with, source.with);
}
assert.equal(job.steps.filter(step => step.uses).length, 2);
const runs = job.steps.filter(step => step.run).map(step => step.run.trim());
for (const command of ['pnpm format:check:rust', 'pnpm lint:rust', 'pnpm test:rust']) assert.ok(runs.includes(command));
assert.ok(runs.includes('pnpm install --frozen-lockfile --ignore-scripts'));
const combined = runs.join('\n');
assert.ok(combined.includes('corepack prepare pnpm@11.15.1 --activate'));
assert.ok(combined.includes('rustup toolchain install 1.97.1'));
assert.ok(combined.includes('rustfmt,clippy'));
assert.ok(combined.includes('sudo apt-get update'));
for (const name of ['libwebkit2gtk-4.1-dev','build-essential','curl','wget','file','libxdo-dev','libssl-dev','libayatana-appindicator3-dev','librsvg2-dev','pkg-config']) assert.ok(combined.includes(name));
const scriptDir = 'tmp/linux-rust-repair/ci-shell';
fs.mkdirSync(scriptDir, {recursive: true});
runs.forEach((script, index) => fs.writeFileSync(`${scriptDir}/step-${index}.sh`, `${script}\n`));
const result = {yamlParsed:true,rootSettingsPreserved:true,existingJobsPreserved:true,onlyNewJob:'ubuntu-native',runner:job['runs-on'],pinnedActions:true,threeRustCommandsMatchPassingComparison:true,shellScripts:runs.length,remoteExecution:false};
fs.writeFileSync('tmp/linux-rust-repair/ci-structure.json', JSON.stringify(result,null,2)+'\n');
console.log(JSON.stringify(result));
